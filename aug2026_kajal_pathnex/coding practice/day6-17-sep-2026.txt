       name: docker
       state: started
       enabled: yes

## Shell Script - check for running services

#!/bin/bash
services=("nginx" "docker" "httpd")
for service in "${services[0]}"
 do 
   if systemctl is-active --quit $service; then
     echo "$service is running"
   else
     echo "$service is not running"
   if 
 done
