#Day-04 Basic CI/CD concepts with jenkins & Gitlab

# Jenkins file - basic CI Pipeline

pipeline {

agent any
stages{
  stage('Build'){
    steps{
     echo 'Building project'
}
}

  stage('Test'){
   steps{
     echo 'Running tests'
}
}

  stage('Deploy'){
    steps{
      echo "Deploying application"
}
}

}

}


## GitLAb CI/CD - Basic pipeline

stages:
 - build
 - test
 - deploy

build:
  stage: build
  script:
    - echo "Building project"

test:
  stage: test
  script:
    - echo "Running project"

Deploy:
  stage: deploy
  script:
    - echo "Deploying application"


## Ansible - Install and Configure nginx
 
- name: Install and Configure nginx
  hosts: all
  become: yes

tasks:
  - name: Install nginx
    yum:
     name: nginx
     stage: present

  - name: configure nginx
    template:
       src: nginx.conf.j2
       dest: /etc/nginx/nginx.conf
 
  - name: Start nginx service
    service:
      name: nginx
      state: started
      enabled: yes
