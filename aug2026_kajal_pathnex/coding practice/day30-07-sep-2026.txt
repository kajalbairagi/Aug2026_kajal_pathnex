##SHell script

#!/bin/bash

while true
do
   echo " SYSTEM MENU "
 echo "1 ) IP"
 echo "2) DISK"
 echo "3) Memory"
 echo "4) Exit"

read -p "Enter your choice: " choice

case $choice in

1)
echo "IP address"
hostname -I
;;

2)
echo "Disk Usage"
df -h 
;;

3)
echo "Memory Usage"
free -h 
;;

4)
echo "Exiting.."
exit 0 
;;

*)
echo "Invalid Choice"
;;


esac
done



## Jenkins

pipeline{
  agent any

  environment{
    INSTITUTE_NAME = "Pathnex"
    TEAM = "Devops"
    ENV  = "Prod"
    PROJECT = "Pathnex- Training"
}

 stages{
   stage('Initialize'){
     steps{
       echo "Institute : ${INSITUTE_NAME}"
       echo "Team : ${TEAM}"
       echo "Environment : ${PROJECT}"
 

 }
}

    stage('Infrastructure'){
      steps {
        echo "Creating EC2 infra.."
    
         dir('terraform'){
          sh 'terraform init'
         
          sh '''
              terraform plan \
              - var= "project= ${PROJECT}"
              '''

          sh '''
              terraform apply - auto-approve \
               - var="project= ${PROJECT}"
              '''

}

}

}

 stage('App Deployment'){

      echo "Configuring server using Ansible..."

}
   echo "Deploying application to kubernetes..."
     dir('kubernetes'){

}
 echo "Checking kubernetes resources..."
  sh 'kubectl get pods'
  sh 'kubectl get svc'
}

}

   stage('Monitoring Checks'){

       chmod +x scripts/monitoring.sh

         '''
}
}
}

post {
   success{

}
 failure{
