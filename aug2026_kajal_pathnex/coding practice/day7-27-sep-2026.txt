 # Day 7 _ automating Cloud Infrastructure with terraform  and Ansible 

## Ansible - Setup nginx with Custom config

---
- name: Setup Nginex with custom config
  hosts: all
  become: yes
  tasks:
    - name: Install nginx
      yum:
       name: nginx
       state: present
   
    - name: Copy custom nginx.conf
      copy:
       src: nginx.conf
       dest: /etc/nginx/nginx.conf

    - name: Start nginx service
      service:
        name: nginx
        state: started
        enabled: yes



## Terraform - provision EC2 with IAm Role

resource "aws_iam_role" "ec2_role"{
  name = "pathnex_ec2_role"
  assume_role_policy = jsonencode({
    version = "2012-10-17"
    Statement = [
        {
          Action = "sts:AssumeRole"
          Effect = "Allow"
          Principal = {
            service = "ec2.amazonaws.com"
}
},
]
})
}

resource "aws_instance" "ec2_instance"{
    ami = "ami-0abcd1234abcd1234"
    instance_type = "t3.micro"
    iam_instance_profile = aws_iam_role.ec2_role.name

tags = {
  Name = "Pathnex-ec2"
}
}
