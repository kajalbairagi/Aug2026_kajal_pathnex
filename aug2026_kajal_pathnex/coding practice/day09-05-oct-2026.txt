# Ansible - Setup Docker Container for Nginx
---
- name: Setup docker container for nginx
  hosts: all
  become: yes
  
  tasks:
   - name: Pull Nginx image
     docker_image:
       name: nginx
       source: pull
   
   - name: Run Nginx container
     docker_image:
       name: nginx-container
       image: nginx
       state: started 
       published_ports:
         - "8080:80"




## Terraform - EC2 with Auto Scaling Group

resource "aws_launch_configuration" "ec2_config"{
   name = "ec2_ASG_config"

   instance_type = "t3.micro"

 resource "aws_auto_scaling_group" "asg_group"{
    desired_capacity = 2
    max_size         = 3
    min_size         = 1
    vpc_zone_identifier = ["subnet-12345677"]
    launch_configuration = aws_launch_configuration.ec2_config.id
}


# Kubernetes - Horizontal Pod Autoscaler (HPA)

apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-deployment
spec:
   replicas: 1
   selector:
     matchlabels:
      app: pathnex-oct
   template:
     metadata:
      labels:
         app: pathnex-oct
     spec:
      containers:
       - name: nginx
         image: nginx
         ports:
          - containerPort: 80

---
 

apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
   name: oathnex-hpa

spec:
  scaleTargetref:
   apiVersion: apps/v1
   kind: Deployment
   name: pathnex-deployment
 minReplicas: 1
 maxReplicas: 10
 targetCPUUtilizationPercentage: 50


## Jenkinsfile - Deploy to k8s

pipeline{
  agent any
  stages{
   stage('Deploy to kubernetes'){
    steps{
       script{
         sh 'kubectl apply -f deployment.yaml'
}
}
}
}
}
