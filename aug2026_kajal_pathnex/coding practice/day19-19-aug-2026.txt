## Ansible - Deploy app using Git 

- name: Deploy app from git for pathnex
  hosts: all
  become: yes

  tasks:
    - name: Install git
      yum:
       name: git
       state: present

    - name: clone repository
      git: 
        repo: "git@github.com:kajalbairagi/Aug2026_kajal_pathnex.git"
        dest: "/opt/pathnex-app"

## Kubernetes - Deployment Rolling Update (Detailed)

apiVersion: apps/v1
kind: Deployment
metadata: 
  name: pathnex-rollout

spec:
  replicas: 3
  strategy:
    type: RollingUpdate
    rollingUpdate:
       maxSurge: 2
       maxUnavailable: 1
  selector:
     matchlabels:
       app: rollout
  template:
    metadata:
      labels:
        app: rollout
    spec:
     containers:
        - name: app
          image: nginx
