# Ansible - nginx.yml

- name: Install and configure nginx
  hosts: all
  become: yes

  tasks:
   - name: Install Nginx
     yum:
     name: nginx
     state: present

   - name: Start and enable Nginx
     service:
        name: nginx
        state: started
        enabled: yes

## Kubernetes  - nginx -deployment. yaml

apiVersion: apps/v1
kind: Deployment
metadata: 
  name: nginx-deployment
 
spec:
  replicas: 3

selector:
  matchLabels:
   app: nginx

template:
   metadata:
   labels:
     app: nginx

spec:
  containers:
    - name: nginx
      image: nginx
      ports:
        - containerPort: 80

## Multi-environment Pipeline

 # Jenkins pipeline - Deploy to Multiple environments

pipeline{
agent any
environment{
  INSTITUTE_NAME = "Pathnex"
}

stages{
  stage('Deploy to Dev'){
    steps {
      sh 'echo " Deploying $INSTITUTE_NAME to dev environment"'
}
}
    stage('Deploy to Staging'){
      steps{
        input message: "Approve staging Deployment"
        sh 'echo "Deployment $INSTITUTE_NAME to Staging environment"'
}

}

    stage('Deployment to prod '){
      steps{
          input message: "Approve prod Deployment?"
            sh 'echo "Deployng $INSTITUTE_NAME to prod environment"'
}

}


}

}
